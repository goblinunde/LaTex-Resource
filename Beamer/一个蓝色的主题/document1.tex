% -----------------------------------------------------------
% 2. 定义深色主题颜色
% -----------------------------------------------------------
\definecolor{editorDark}{RGB}{40, 44, 52}     % 代码背景深灰色
\definecolor{editorFrame}{RGB}{33, 37, 43}    % 边框更深
\definecolor{editorTitle}{RGB}{171, 178, 191} % 标题文字灰白色
\definecolor{keywordBlue}{RGB}{97, 175, 239}  % 关键字亮蓝
\definecolor{cmdPurple}{RGB}{198, 120, 221}   % 命令亮紫
\definecolor{strGreen}{RGB}{152, 195, 121}    % 字符串亮绿

% -----------------------------------------------------------
% 3. 配置深色模式代码高亮 (Dark Mode Syntax Highlighting)
% -----------------------------------------------------------
\lstdefinestyle{darkstyle}{
	language={[LaTeX]TeX},
	basicstyle=\ttfamily\small\color{white},
	keywordstyle=\color{keywordBlue}\bfseries,
	commentstyle=\color{gray!60}\itshape,
	texcsstyle=*\color{cmdPurple},
	morekeywords={begin, end, section, subsection, textbf, textit, underline, mathbb, mathcal},
	numbers=left,
	numberstyle=\tiny\color{gray!50},
	frame=none,
	breaklines=true,
	columns=flexible,
	keepspaces=true,
	tabsize=6,
	showtabs=false
}

% -----------------------------------------------------------
% 4. 设计 macOS 窗口风格的盒子
% -----------------------------------------------------------
\tcbset{
	macos-editor/.style={
		skin=bicolor,
		colframe=editorFrame,
		colback=editorDark,
		colbacklower=white,
		coltitle=editorTitle,
		fonttitle=\sffamily\bfseries\small,
		arc=6pt,
		boxrule=1pt,
		drop fuzzy shadow,
		title style={opacity=0},
		top=16pt, bottom=10pt,
		overlay={
			\fill[editorFrame] (frame.north west) rectangle ([yshift=-24pt]frame.north east);
			\fill[red!60] ([xshift=15pt, yshift=-12pt]frame.north west) circle (3pt);
			\fill[orange!60] ([xshift=28pt, yshift=-12pt]frame.north west) circle (3pt);
			\fill[green!60] ([xshift=41pt, yshift=-12pt]frame.north west) circle (3pt);
		},
		attach boxed title to top center,
		boxed title style={empty},
	}
}

% -----------------------------------------------------------
% 5. 定义环境：左代码，右结果 (Dark Mode)
% -----------------------------------------------------------
\newtcblisting{code-left}[2][]{
	macos-editor,
	listing style=darkstyle,
	sidebyside,
	listing and text,
	lefthand width=0.45\textwidth,
	valign=center,
	title={#2},
	#1
}

% -----------------------------------------------------------
% 6. 定义环境：上代码，下结果 (Dark Mode)
% -----------------------------------------------------------
\newtcblisting{code-top}[2][]{
	macos-editor,
	listing style=darkstyle,
	title={#2},
	#1
}
